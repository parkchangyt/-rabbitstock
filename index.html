export default { async fetch(request, env) { const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } const url = new URL(request.url); const path = url.pathname; if (path === "/") { return json({ ok: true, service: "RabbitStock API", version: "2.0" }, cors); } if (path === "/quote") { const symbol = (url.searchParams.get("symbol") || "BRNX") .trim() .toUpperCase(); if (!env.TWELVE_API_KEY) { return json({ ok: false, error: "TWELVE_API_KEY 없음" }, cors, 500); } try { const api = "https://api.twelvedata.com/quote" + "?symbol=" + encodeURIComponent(symbol) + "&apikey=" + encodeURIComponent(env.TWELVE_API_KEY); const r = await fetch(api); const data = await r.json(); if (data.status === "error") { return json({ ok: false, error: data.message || "Twelve Data 오류" }, cors, 400); } return json({ ok: true, symbol: symbol, data: data }, cors); } catch (e) { return json({ ok: false, error: "Twelve Data 연결 실패", detail: String(e) }, cors, 502); } } return json({ ok: false, error: "Not Found" }, cors, 404); } }; function json(data, cors, status = 200) { return new Response( JSON.stringify(data), { status: status, headers: { ...cors, "Content-Type": "application/json; charset=UTF-8" } } ); }
